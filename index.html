// Smart Native App Deep Linking for Mobile Devices (Android Intent URI & iOS custom schemes) function setupMobileAppDeepLinks() { const ua = navigator.userAgent || ''; const isAndroid = /Android/i.test(ua); const isIOS = /iPhone|iPad|iPod/i.test(ua); const appLinks = document.querySelectorAll('.app-link'); if (isAndroid) { // Android Chrome enforces direct user navigation on official intent:// URIs // Package targeting triggers Android OS Package Manager to launch installed apps directly appLinks.forEach(link => { const androidIntent = link.getAttribute('data-android-intent'); if (androidIntent) { link.setAttribute('href', androidIntent); link.removeAttribute('target'); // Opens inside intent handler without creating dead browser tabs } }); return; } if (isIOS) { // iOS Safari custom scheme handler with timed fallback appLinks.forEach(link => { link.addEventListener('click', (e) => { const appScheme = link.getAttribute('data-app-scheme'); const webUrl = link.getAttribute('data-web-url') || link.href; if (appScheme) { e.preventDefault(); let hasLeftPage = false; const onVisibilityChange = () => { if (document.hidden || document.webkitHidden) { hasLeftPage = true; } }; document.addEventListener('visibilitychange', onVisibilityChange); window.location.href = appScheme; setTimeout(() => { document.removeEventListener('visibilitychange', onVisibilityChange); if (!hasLeftPage && !document.hidden && !document.webkitHidden) { window.location.href = webUrl; } }, 1500); } }); }); } // On desktop, elements keep target="_blank" and their original web URLs } window.onload = function() { renderBookmarks(); setupMobileAppDeepLinks(); updateMarketStatus(); updateExtendedHoursStatus(); setInterval(updateMarketStatus, 1000); setInterval(updateExtendedHoursStatus, 1000); fetchLiveBenchmarkRates(); setInterval(fetchLiveBenchmarkRates, 15000); };
